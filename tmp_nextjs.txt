"use client";
import { Button } from "@/components/ui/button";
import { isEmpty } from "@/utils/utils";
import { useRouter } from "next/navigation";
import { useEffect, useState } from "react";
import { toast } from "../ui/use-toast";
import {
  FieldValue,
  collection,
  doc,
  getDoc,
  getDocs,
  increment,
  query,
  serverTimestamp,
  setDoc,
  where,
} from "firebase/firestore";
import { db } from "@/firebase/firebase";
import { Progress } from "@/components/ui/progress";

export function Hero() {
  const router = useRouter();

  const [email, setEmail] = useState("");
  const [progress, setProgress] = useState(0);
  const maxPosti = 2000;

  const fetchData = async () => {
    const docRef = doc(db, "Contatti app", "Data");
    const docSnap = await getDoc(docRef);

    if (docSnap.exists()) {
      const docData = docSnap.data();
      return docData;
    } else {
      toast({
        variant: "destructive",
        title: "Errore.",
        description: "Errore interno. Riprova più tardi.",
      });
    }
  };

  const handleInvia = async (ev: any) => {
    ev.preventDefault();

    if (isEmpty(email)) {
      toast({
        variant: "destructive",
        title: "Errore.",
        description: "Controlla il campo email e riprova.",
      });
      return;
    }

    const emailSanitized = email.toLowerCase();

    const q = query(
      collection(db, "Contatti app"),
      where("email", "==", emailSanitized)
    );
    const querySnapshot = await getDocs(q);
    if (!querySnapshot.empty) {
      toast({
        variant: "destructive",
        title: "Email già reigstrata.",
        description:
          "Questa email è già stata inserita nella lista dei futuri beta tester.",
      });
      setEmail("");
      return;
    }

    fetchData()
      .then(async (docData) => {
        if (docData!.postiDisponibili === 0) {
          toast({
            variant: "destructive",
            title: "Posti terminati.",
            description:
              "Seguici sui nostri social per scoprire quando apriremo nuovi posti.",
          });
          setEmail("");
          return;
        }

        await setDoc(doc(db, "Contatti app", Date.now().toString()), {
          email: emailSanitized,
          timestamp: serverTimestamp(),
        });

        await setDoc(
          doc(db, "Contatti app", "Data"),
          {
            postiDisponibili: increment(-1),
          },
          { merge: true }
        );

        setEmail("");
        setProgress(progress + 1);

        toast({
          variant: "success",
          title: "Pre-registrazione completata.",
          description:
            "Sei stato inserito nella lista dei futuri beta tester 🍾.",
        });
      })
      .catch(() => {
        toast({
          variant: "destructive",
          title: "Errore.",
          description: "Errore interno. Riprova più tardi.",
        });
      });
  };

  useEffect(() => {
    fetchData()
      .then((docData) => {
        const postiOccupati = maxPosti - docData!.postiDisponibili;

        const timer = setTimeout(() => setProgress(postiOccupati), 500);
        return () => clearTimeout(timer);
      })
      .catch((err) => {
        toast({
          variant: "destructive",
          title: "Errore.",
          description: "Errore interno. Riprova più tardi.",
        });
      });
  }, [progress]);

  return (
    <section className="body-font lg:p-20">
      <div className="container mx-auto flex px-5 py-24 md:flex-row flex-col items-center">
        <div className="lg:flex-grow md:w-1/2 flex md:pr-10 flex-col md:items-start md:text-left mb-16 md:mb-0 items-center text-center">
          <h1 className="max-w-2xl mb-4 text-4xl font-extrabold tracking-tight leading-none md:text-5xl xl:text-6xl dark:text-white text-center md:text-left">
            Vuoi essere tra i primi utilizzatori dell’app?
          </h1>
          <p className="mb-8 text-base md:text-lg lg:text-xl leading-relaxed w-full lg:w-4/5">
            Ci sono {maxPosti} posti disponibili: pre-registrati lasciando la tua email
            qui sotto e ti riserveremo un posto!
          </p>

          <form
            className="flex w-full md:justify-start justify-center items-center"
            onSubmit={(ev) => handleInvia(ev)}
          >
            <div className="relative mr-4 w-full sm:w-1/2 md:w-full lg:w-1/2">
              <input
                type="email"
                id="email"
                name="email"
                className="block p-3 w-full text-sm text-gray-900 bg-gray-50 rounded-lg border border-gray-300 shadow-sm focus:ring-primary-500 focus:border-primary-500 dark:bg-gray-700 dark:border-gray-600 dark:placeholder-gray-400 dark:text-white dark:focus:ring-primary-500 dark:focus:border-primary-500 dark:shadow-sm-light"
                placeholder="prova@esempio.it"
                required
                value={email}
                onChange={(ev) => setEmail(ev.target.value)}
                pattern="[a-zA-Z0-9\.]+@[a-z]+\.[a-z]+"
              />
            </div>
            <Button variant="default">Invia</Button>
          </form>

          <div className="flex mt-6 w-full md:justify-start justify-center items-center">
            <Progress
              value={progress}
              className="relative mr-4 w-full sm:w-1/2 md:w-full lg:w-1/2"
              max={maxPosti}
            />
            <span className="text-sm sm:text-base">{`${progress} / ${maxPosti} partecipano`}</span>
          </div>
        </div>
        <div className="lg:mt-0 lg:col-span-5 lg:flex mx-auto">
          <img
            src="/images/screen/1.png"
            alt="mockup"
            width={250}
            height={100}
          />
        </div>
      </div>
    </section>
  );
}
export default Hero;

"use client";

import React, { useRef, useState } from "react";
import { Button } from "../../components/ui/button";
import emailjs from "@emailjs/browser";
import { useToast } from "@/components/ui/use-toast";
import { isEmpty, isValidEmail } from "@/utils/utils";

export function ContactForm() {
  const form = useRef<any>();
  const { toast } = useToast();

  const [nome, setNome] = useState("");
  const [cognome, setCognome] = useState("");
  const [email, setEmail] = useState("");
  const [oggetto, setOggetto] = useState("");
  const [messaggio, setMessaggio] = useState("");

  const checkFields = () => {
    if (
      isEmpty(nome) ||
      isEmpty(cognome) ||
      isEmpty(email) ||
      isEmpty(oggetto) ||
      isEmpty(messaggio)
    ) {
      return false;
    }
    return true;
  };

  const sendEmail = (e: any) => {
    e.preventDefault();

    if (!isValidEmail(email)) {
      toast({
        variant: "destructive",
        title: "Errore.",
        description: "Controlla il campo email e riprova.",
      });
      return;
    }

    if (checkFields() === true) {
      emailjs
        .sendForm(
          "service_mj3xrmc",
          "template_aojje4e",
          form.current!,
          "umAdQk5WuCS_gPHDx"
        )
        .then(
          (result) => {
            toast({
              variant: "success",
              title: "Messaggio inviato.",
              description:
                "Grazie per averci contattato. Ti risponderemo il prima possibile.",
            });
          },
          (error) => {
            toast({
              variant: "destructive",
              title: "Errore.",
              description: "Errore nell'invio dell'email. Riprova più tardi.",
            });
          }
        );

      setNome("");
      setCognome("");
      setEmail("");
      setOggetto("");
      setMessaggio("");
    } else {
      toast({
        variant: "destructive",
        title: "Errore.",
        description: "Controlla i campi vuoti e riprova.",
      });
    }
  };

  return (
    <section className="bg-white dark:bg-gray-900 mt-16">
      <div className="py-8 lg:py-16 px-4 mx-auto max-w-screen-md">
        <h2 className="mb-4 text-4xl tracking-tight font-extrabold text-center text-gray-900 dark:text-white">
          CONTATTACI
        </h2>
        <p className="mb-8 lg:mb-16 font-light text-center text-gray-500 dark:text-gray-400 sm:text-xl">
          Sei un club e vuoi essere presente sull’app? <br/>Compila
          il modulo e ti risponderemo al più presto.
        </p>
        <form className="space-y-8" ref={form}>
          <div>
            <label
              htmlFor="name"
              className="block mb-2 text-sm font-medium text-gray-900 dark:text-gray-300"
            >
              Nome
            </label>
            <input
              type="text"
              id="name"
              name="name"
              className="block p-3 w-full text-sm text-gray-900 bg-gray-50 rounded-lg border border-gray-300 shadow-sm focus:ring-primary-500 focus:border-primary-500 dark:bg-gray-700 dark:border-gray-600 dark:placeholder-gray-400 dark:text-white dark:focus:ring-primary-500 dark:focus:border-primary-500 dark:shadow-sm-light"
              placeholder="Mario"
              required
              value={nome}
              onChange={(ev) => setNome(ev.target.value)}
            />
          </div>
          <div>
            <label
              htmlFor="surname"
              className="block mb-2 text-sm font-medium text-gray-900 dark:text-gray-300"
            >
              Cognome
            </label>
            <input
              type="text"
              id="surname"
              name="surname"
              className="block p-3 w-full text-sm text-gray-900 bg-gray-50 rounded-lg border border-gray-300 shadow-sm focus:ring-primary-500 focus:border-primary-500 dark:bg-gray-700 dark:border-gray-600 dark:placeholder-gray-400 dark:text-white dark:focus:ring-primary-500 dark:focus:border-primary-500 dark:shadow-sm-light"
              placeholder="Rossi"
              required
              value={cognome}
              onChange={(ev) => setCognome(ev.target.value)}
            />
          </div>
          <div>
            <label
              htmlFor="email"
              className="block mb-2 text-sm font-medium text-gray-900 dark:text-gray-300"
            >
              Email
            </label>
            <input
              type="email"
              id="email"
              name="email"
              className="shadow-sm bg-gray-50 border border-gray-300 text-gray-900 text-sm rounded-lg focus:ring-primary-500 focus:border-primary-500 block w-full p-2.5 dark:bg-gray-700 dark:border-gray-600 dark:placeholder-gray-400 dark:text-white dark:focus:ring-primary-500 dark:focus:border-primary-500 dark:shadow-sm-light"
              placeholder="prova@esempio.com"
              required
              value={email}
              onChange={(ev) => setEmail(ev.target.value)}
            />
          </div>
          <div>
            <label
              htmlFor="subject"
              className="block mb-2 text-sm font-medium text-gray-900 dark:text-gray-300"
            >
              Oggetto
            </label>
            <input
              type="text"
              id="subject"
              name="subject"
              className="block p-3 w-full text-sm text-gray-900 bg-gray-50 rounded-lg border border-gray-300 shadow-sm focus:ring-primary-500 focus:border-primary-500 dark:bg-gray-700 dark:border-gray-600 dark:placeholder-gray-400 dark:text-white dark:focus:ring-primary-500 dark:focus:border-primary-500 dark:shadow-sm-light"
              placeholder="Facci sapere come possiamo aiutarti"
              required
              value={oggetto}
              onChange={(ev) => setOggetto(ev.target.value)}
            />
          </div>
          <div className="sm:col-span-2">
            <label
              htmlFor="message"
              className="block mb-2 text-sm font-medium text-gray-900 dark:text-gray-400"
            >
              Messaggio
            </label>
            <textarea
              id="message"
              name="message"
              rows={6}
              className="block p-2.5 w-full text-sm text-gray-900 bg-gray-50 rounded-lg shadow-sm border border-gray-300 focus:ring-primary-500 focus:border-primary-500 dark:bg-gray-700 dark:border-gray-600 dark:placeholder-gray-400 dark:text-white dark:focus:ring-primary-500 dark:focus:border-primary-500"
              placeholder="Scrivi qui il tuo messaggio..."
              value={messaggio}
              onChange={(ev) => setMessaggio(ev.target.value)}
            ></textarea>
          </div>
          <Button
            type="submit"
            variant="default"
            onClick={(ev) => sendEmail(ev)}
          >
            Invia
          </Button>
        </form>
      </div>
    </section>
  );
}
export default ContactForm;
